cmake_minimum_required(VERSION 3.22)
project(paper_pico_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(paper_core STATIC common/shared_sample.cpp transport/jpg0_parser.cpp transport/raw_jpeg_parser.cpp calibration/calibration.cpp calibration/gaze_estimator.cpp calibration/calibration_store.cpp calibration/calibration_fit.cpp vision/pupil_detector.cpp vision/jpeg_decoder.cpp)
target_include_directories(paper_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/common ${CMAKE_CURRENT_SOURCE_DIR}/transport ${CMAKE_CURRENT_SOURCE_DIR}/calibration ${CMAKE_CURRENT_SOURCE_DIR}/vision)
add_executable(paper_bridge_daemon daemon/main.cpp daemon/paper_usb.cpp daemon/sample_publisher.cpp)
target_include_directories(paper_bridge_daemon PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/daemon ${CMAKE_CURRENT_SOURCE_DIR}/calibration)
target_link_libraries(paper_bridge_daemon PRIVATE paper_core)
add_executable(calibrate_csv tools/calibrate_csv.cpp)
target_include_directories(calibrate_csv PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/calibration)
target_link_libraries(calibrate_csv PRIVATE paper_core)
add_executable(paper_core_tests tests/test_core.cpp hook/eye_hook_controller.cpp hook/eye_output_adapter.cpp)
target_include_directories(paper_core_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/calibration ${CMAKE_CURRENT_SOURCE_DIR}/common ${CMAKE_CURRENT_SOURCE_DIR}/hook)
target_link_libraries(paper_core_tests PRIVATE paper_core)
add_library(paper_bridge_hook_stub SHARED hook/zygisk_module.cpp hook/runtime_resolver.cpp hook/runtime_probe.cpp hook/eye_output_adapter.cpp hook/shared_sample_reader.cpp hook/inline_hook.cpp hook/eye_hook_controller.cpp common/shared_sample.cpp)
target_include_directories(paper_bridge_hook_stub PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/hook ${CMAKE_CURRENT_SOURCE_DIR}/common)
target_link_libraries(paper_bridge_hook_stub PRIVATE log)
enable_testing()
add_test(NAME paper_core_tests COMMAND paper_core_tests)
